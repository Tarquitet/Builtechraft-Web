<WikiRule num="8.1" title="Comercio dentro del servidor" sub="Entre jugadores = libre." severity="leve" scope="juego">

<WikiRuleItem code="8.1.a">Libre intercambio de ítems entre jugadores dentro del servidor.</WikiRuleItem>

<WikiRuleItem code="8.1.b">Tiendas con carteles bien señalizados son bienvenidas.</WikiRuleItem>

</WikiRule>

<WikiRule num="8.2" title="Estafas y dinero real" sub="Línea roja." severity="expulsion" scope="ambos">

<WikiRuleItem code="8.2.a">
  **Prohibido vender ítems, cuentas o servicios por dinero real** (PayPal, cripto, etc.). Expulsión directa.
</WikiRuleItem>

<WikiRuleItem code="8.2.b">
  Estafar a otros jugadores (prometer algo y no cumplir) = sanción según gravedad.
</WikiRuleItem>

<WikiRuleItem code="8.2.c">No uses el servidor para promover ventas externas o negocios personales.</WikiRuleItem>

</WikiRule>

<WikiRule num="8.3" title="Competencia desleal" sub="En zonas comerciales." severity="leve" scope="juego">

<WikiRuleItem code="8.3.a">No copies diseños de tiendas ajenas para socavar precios deliberadamente.</WikiRuleItem>

<WikiRuleItem code="8.3.b">Respeta la economía del servidor: no infles precios artificialmente.</WikiRuleItem>

</WikiRule>
