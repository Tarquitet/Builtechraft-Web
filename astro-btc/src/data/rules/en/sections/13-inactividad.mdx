import WikiRuleItem from '@/components/wiki/WikiRuleItem.astro';

<WikiRule num="13.1" title="Your account and builds" sub="We do not remove your work without notice." severity="leve" scope="juego">

<WikiRuleItem code="13.1.a">
  **Inactivity does not delete your account:** your builds and items remain intact even if you are away for months.
</WikiRuleItem>

<WikiRuleItem code="13.1.b">
  **Your whitelist remains active:** inactivity alone will not remove you from the server.
</WikiRuleItem>

<WikiRuleItem code="13.1.c">
  **Let us know in advance:** if you expect to be away for a long time (vacation, studies, or work), tell us through
  Discord or a ticket. This helps us plan.
</WikiRuleItem>

</WikiRule>

<WikiRule num="13.2" title="Shops and commercial areas" sub="We keep space available for active players." severity="leve" scope="juego">

<WikiRuleItem code="13.2.a">
  **After 21 days of inactivity** without prior notice, staff may remove your shop.
</WikiRuleItem>

<WikiRuleItem code="13.2.b">
  **To extend the deadline:** open a ticket and tell us how long you expect to be away.
</WikiRuleItem>

<WikiRuleItem code="13.2.c">
  Staff will store items from the shop for an additional 30 days before releasing them.
</WikiRuleItem>

</WikiRule>

<WikiRule num="13.3" title="Banned players" sub="Temporary and permanent bans are handled differently." severity="leve" scope="juego">

<WikiRuleItem code="13.3.a">
  **Temporary ban:** your account and builds remain intact during the sanction (rules 13.1 and 13.2 still apply).
</WikiRuleItem>

<WikiRuleItem code="13.3.b">
  **Permanent ban:** your items will be removed and the area released once the final removal is confirmed.
</WikiRuleItem>

</WikiRule>
