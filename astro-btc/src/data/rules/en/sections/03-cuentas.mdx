<WikiRule num="3.2" title="Your account, your responsibility" sub="No exceptions." severity="grave" scope="ambos">

<WikiRuleItem code="3.2.a">
  If a friend, family member, or stranger uses your account to break the rules, the sanction applies to your account.
</WikiRuleItem>

<WikiRuleItem code="3.2.b">
  Keep your password secure. Staff will not remove sanctions caused by poor account security.
</WikiRuleItem>

<WikiRuleItem code="3.2.c">

</WikiRuleItem>

</WikiRule>

<WikiRule num="3.3" title="Alternate accounts and appeals" sub="A ban cannot be bypassed." severity="expulsion" scope="ambos">

<WikiRuleItem code="3.3.a">

</WikiRuleItem>

<WikiRuleItem code="3.3.b">
  Alternate accounts are allowed only with explicit staff approval (for example, after losing access to an account).
</WikiRuleItem>

<WikiRuleItem code="3.3.c">
