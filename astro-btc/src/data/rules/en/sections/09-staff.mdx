import WikiRuleItem from '@/components/wiki/WikiRuleItem.astro';

<WikiRule num="9.1" title="Working with staff" sub="We are volunteers, not employees." severity="grave" scope="ambos">

<WikiRuleItem code="9.1.a">
  Be patient when waiting for a response. We cannot provide immediate attention: we work, study, and rest.
</WikiRuleItem>

<WikiRuleItem code="9.1.b">
  Repeatedly mentioning staff without a real emergency may result in a temporary mute.
</WikiRuleItem>

<WikiRuleItem code="9.1.c">Insulting or belittling a team member may result in your access being revoked.</WikiRuleItem>

</WikiRule>

<WikiRule num="9.2" title="Mistakes and appeals" sub="We are human." severity="leve" scope="ambos">

<WikiRuleItem code="9.2.a">
  If an admin makes a mistake, report it calmly and respectfully through a support ticket. It will be investigated.
</WikiRuleItem>

<WikiRuleItem code="9.2.b">
  Anyone who has been banned may appeal through a ticket. Staff make the final decision.
</WikiRuleItem>

</WikiRule>

<WikiRule num="9.3" title="Sanction system" sub="Severity levels are transparent." severity="leve" scope="ambos">

<WikiRuleItem code="9.3.a">**Minor:** Warning or temporary mute. For minor or first-time offenses.</WikiRuleItem>

<WikiRuleItem code="9.3.b">**Serious:** Temporary ban. For repeated or moderately harmful offenses.</WikiRuleItem>

<WikiRuleItem code="9.3.c">
  **Removal:** Permanent ban. For serious offenses (X-Ray, scams, ban evasion, or impersonation).
</WikiRuleItem>

<WikiRuleItem code="9.3.d">Staff consider context, intent, impact, and repeat offenses when deciding.</WikiRuleItem>

</WikiRule>

<WikiRule num="9.4" title="Discord roles" sub="Do not request them." severity="leve" scope="discord">

<WikiRuleItem code="9.4.a">
  Do not ask for roles. They are assigned only when staff needs a new team member.
</WikiRuleItem>

</WikiRule>

<WikiRule num="9.5" title="Consistent enforcement" sub="A rule that is selectively enforced loses authority." severity="grave" scope="ambos">

<WikiRuleItem code="9.5.a">
  Staff applies the rules equally to everyone. If a rule should no longer exist, staff changes the rule before changing
  how it is enforced.
</WikiRuleItem>

<WikiRuleItem code="9.5.b">
  Looking the other way is also a decision: staff is responsible for what it allows in spaces where minors are present.
</WikiRuleItem>

</WikiRule>
