import WikiRuleItem from '@/components/wiki/WikiRuleItem.astro';

<WikiRule num="5.1" title="Hacks and X-Ray" sub="Permanent ban, no exceptions." severity="expulsion" scope="juego">

<WikiRuleItem code="5.1.a">
  **X-Ray results in a permanent ban.** This includes mods, resource packs, transparent lava, or using the world seed to
  find ores.
</WikiRuleItem>

<WikiRuleItem code="5.1.b">
  Any mod that provides an unfair advantage (fly, reach, killaura, ESP, tracers) is prohibited.
</WikiRuleItem>

<WikiRuleItem code="5.1.c">
  Using the seed to locate structures (including copper or iron veins) is also considered X-Ray.
</WikiRuleItem>

</WikiRule>

<WikiRule num="5.2" title="Automation" sub="Do not play for yourself." severity="expulsion" scope="juego">

<WikiRuleItem code="5.2.a">
  Prohibited: autoclickers, auto-mining, auto-attacking, auto-eating, auto-replanting, and macros.
</WikiRuleItem>

<WikiRuleItem code="5.2.b">Authorized fake players (`/player`) are the only controlled exception.</WikiRuleItem>

<WikiRuleItem code="5.2.c">
  Exploiting vanilla or plugin bugs for an advantage may result in sanctions. Report bugs instead of using them.
</WikiRuleItem>

</WikiRule>

<WikiRule num="5.3" title="PvP and combat" sub="Only with consent." severity="grave" scope="juego">

<WikiRuleItem code="5.3.a">
  PvP is allowed only when both players consent. Return the items to the player who died.
</WikiRuleItem>

<WikiRuleItem code="5.3.b">Do not repeatedly kill the same player. Give them time to disable PvP.</WikiRuleItem>

<WikiRuleItem code="5.3.c">Do not attack staff while they are carrying out their duties.</WikiRuleItem>

</WikiRule>

<WikiRule num="5.4" title="Gambling" sub="No betting." severity="grave" scope="juego">

<WikiRuleItem code="5.4.a">Chance-based machines (roulette, slots, betting) are prohibited.</WikiRuleItem>

<WikiRuleItem code="5.4.b">Skill-based machines (target shooting, races) are allowed.</WikiRuleItem>

</WikiRule>
