import WikiRuleItem from '@/components/wiki/WikiRuleItem.astro';

<WikiRule num="1.1" title="Respect for everyone" sub="On and off the server." severity="expulsion" scope="ambos">

<WikiRuleItem code="1.1.a">
  Discrimination, harassment, insults, or sharing another person's private information are not tolerated. This includes
  xenophobia, racism, discrimination based on orientation or gender, political harassment, and religious discrimination.
</WikiRuleItem>

<WikiRuleItem code="1.1.b">
  Avoid abusive language and toxic behavior. Do not bring outside controversies or news into the server.
</WikiRuleItem>

<WikiRuleItem code="1.1.c">
  Public drama and fights may result in sanctions. Resolve issues through a support ticket.
</WikiRuleItem>

</WikiRule>

<WikiRule num="1.2" title="Protecting minors" sub="Players of all ages are part of this server." severity="expulsion" scope="ambos">

<WikiRuleItem code="1.2.a">
  **Adult, NSFW, or sexual content results in immediate removal.** This includes builds, messages, skins, and profile
  pictures.
</WikiRuleItem>

<WikiRuleItem code="1.2.b">
  Any sexual solicitation or grooming of a minor results in removal and a report to the appropriate authorities.
</WikiRuleItem>

<WikiRuleItem code="1.2.c">
  Do not share or request personal information from minors, including their real age, location, or private social
  accounts.
</WikiRuleItem>

<WikiRuleItem code="1.2.d">
  Protection is based on age, not perceived maturity. A mature minor may earn trust and responsibilities, but is never
  exempt from protection.
</WikiRuleItem>

</WikiRule>

<WikiRule num="1.3" title="Symbols and inappropriate content" sub="Applies to builds, skins, banners, and profiles." severity="grave" scope="ambos">

<WikiRuleItem code="1.3.a">
  **Prohibited:** political flags, ideologies, controversial symbols (including swastikas and hate symbols), gore,
  violent content, and offensive material.
</WikiRuleItem>

<WikiRuleItem code="1.3.b">
  **Historical builds are allowed:** castles, the Taj Mahal, and monuments are welcome as long as they do not include
  offensive symbols.
</WikiRuleItem>

<WikiRuleItem code="1.3.c">
  This applies equally in Minecraft (builds, skins, capes, and signs) and on Discord (profile pictures, banners, names,
  and bios).
</WikiRuleItem>

<WikiRuleItem code="1.3.d">
  If a staff member asks you to change a symbol, skin, or profile, you must do so. This is about living together
  respectfully, not censorship.
</WikiRuleItem>

<WikiRuleItem code="1.3.e">
  Threatening to leave to avoid a rule does not exempt you from following it. The rules apply for as long as you are
  part of the community.
</WikiRuleItem>

</WikiRule>

<WikiRule num="1.4" title="Chat and communication" sub="The same rules apply in every channel." severity="grave" scope="ambos">

<WikiRuleItem code="1.4.a">
  These rules also apply to private messages. Insulting someone by DM can still result in sanctions.
</WikiRuleItem>

<WikiRuleItem code="1.4.b">Do not spam links to other servers or flood chat with repetitive messages.</WikiRuleItem>

<WikiRuleItem code="1.4.c">
  Do not overuse capital letters or repeatedly mention staff without a real emergency.
</WikiRuleItem>

<WikiRuleItem code="1.4.d">
  Taking part in raids against the community results in permanent removal and an IP ban from the server.
</WikiRuleItem>

</WikiRule>

<WikiRule num="1.5" title="If you use Discord" sub="Discord support is optional, but these rules apply if you join." severity="leve" scope="discord">

<WikiRuleItem code="1.5.a">
  Read each channel's name and description before posting. Do not disrupt voice or text channels.
</WikiRuleItem>

<WikiRuleItem code="1.5.b">Discord's Terms of Service apply throughout the community.</WikiRuleItem>

</WikiRule>
